```C++
#include <iostream>
using namespace std;
int main() {
    unsigned int n = 12345678;
    cout << hex << n << endl; // 输出bc614e
    unsigned char* v = (unsigned char*)(&n);

    // 显示形式有很多种，但是存储形式只有二进制一种 
    cout << v[0]; // 输出N, 78 - ASCII的N
    cout << (int)v[0]; // 输出78
    cout << hex << (int)v[0]; // 输出4e, 78 - 十六进制4e

    // 输出4e 61 bc 0
    // hex 流式输出
    cout << hex << (int)v[0] << " " << (int)v[1] << " " << (int)v[2] << " " << (int)v[3];
    
    // 用printf输出
    /* 格式控制符	含义
        %x	    按小写十六进制输出整数
        %X	    按大写十六进制输出整数
        %02x	按十六进制输出，至少占 2 位，不足时在前面补 0
        %08x	按十六进制输出，至少占 8 位，不足时在前面补 0
        %#x	    输出带 0x 前缀的十六进制数
    */
    printf("%02x %02x %02x %02x", (unsigned int)v[0], (unsigned int)v[1],
                                  (unsigned int)v[2], (unsigned int)v[3]);
    return 0;
}


// 写复杂了
#include <iostream>
#include <string>
using namespace std;
int main() {
    unsigned int n; cin >> n;
    // 因为后修改了n，所有地址里的内容也没有了，后面输出每个字节也没有东西了
    unsigned int t = n;
    unsigned char* v = (unsigned char*)&t;
    char s[9];
    int idx = 0;
    while (n != 0) {
        int c = n % 16;
        if (10 <= c && c <= 15) s[idx++] = 'a' + c - 10;
        else s[idx++] = '0' + c;
        n /= 16;
    }
    // char s[]字符串尾巴加上'\0'，不++，以免把\0也翻转了
    s[idx] = '\0';
    // reverse s
    int l = 0, r = idx-1;
    while (l < r) {
        char t = s[r];
        s[r] = s[l];
        s[l] = t;
        l++, r--;
    }
    cout << s << endl;
    cout << hex << (unsigned int)v[0] << " " << hex << (unsigned int)v[1] << " " << hex << (unsigned int)v[2] <<" " << hex << (unsigned int)v[3];
    return 0;
}
```
