数据存储在内存地址的形式
小端（现在常见）：数据低位存在内存低位
大端：数据高位存在内存低位

```C++
#include <iostream>
#include <string>
using namespace std;
int main() {
    unsigned int n; cin >> n;
    // 因为后修改了n，所有地址里的内容也没有了，后面输出每个字节也没有东西了
    unsigned int t = n;
    unsigned char* v = (unsigned char*)&t;
    char s[9];
    int idx = 0;
    while (n != 0) {
        int c = n % 16;
        if (10 <= c && c <= 15) s[idx++] = 'a' + c - 10;
        else s[idx++] = '0' + c;
        n /= 16;
    }
    // char s[]字符串尾巴加上'\0'，不++，以免把\0也翻转了
    s[idx] = '\0';
    // reverse s
    int l = 0, r = idx-1;
    while (l < r) {
        char t = s[r];
        s[r] = s[l];
        s[l] = t;
        l++, r--;
    }
    cout << s << endl;
    cout << hex << (unsigned int)v[0] << " " << hex << (unsigned int)v[1] << " " << hex << (unsigned int)v[2] <<" " << hex << (unsigned int)v[3];
    return 0;
}
```
