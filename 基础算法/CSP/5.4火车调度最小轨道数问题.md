time complexity: O(n^2)
```c++
#include <iostream>
#include <queue>
#include <stack>
#include <vector>
#include <climits>
using namespace std;
int main() {
    int N; cin >> N;
    queue<int> q;
    for (int i = 0; i < N; i++) {
        int t; cin >> t;
        q.push(t);
    }

    vector<stack<int>> ways;
    while (!q.empty()) {
        // 将队列出来的元素定义为x
        int x = q.front();
        q.pop();
        // 总结：两个变量解决需要满足的两个条件：一栈顶元素小于x,二所有满足一条件的中最小的栈顶元素
        // 初始轨道定义为-1，最小的栈顶定义为最大整数
        int pos = -1;
        int bestTop = INT_MAX;
        // 遍历轨道，找到栈顶最小的轨道，每个轨道是一个栈
        for (int j = 0; j < ways.size(); j++) { // ways最开始的大小我们是不知道的
            if (!ways[j].empty() && 
                ways[j].top() > x && 
                ways[j].top() < bestTop) {
                bestTop = ways[j].top();
                pos = j;
            }
        }
        // 找到了
        if (pos != -1) {
            ways[pos].push(x);
        }
        // 当所有轨道没有小于x的栈顶元素时，增加轨道并把x放入新轨道
        else {
            ways.push_back(stack<int>());
            ways.back().push(x);
        }
    }
    cout << ways.size() << endl;
    return 0;
}```


lower_bound upper_bound基本功能
在已排序的区间 [first, last) 中，upper_bound 用于查找第一个大于（>）给定值 val 的元素，并返回该元素的迭代器（或地址）。如果所有元素都小于或等于 val，则返回 last 位置。

函数名	        升序数组中的查找条件	            未找到时的返回值
lower_bound	    第一个大于或等于（≥）val 的元素	    last
upper_bound	    第一个严格大于（>）val 的元素	    last

vector<int> v = {1, 2, 4, 4, 4, 7, 15};
