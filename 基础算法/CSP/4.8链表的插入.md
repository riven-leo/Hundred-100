循环里出现负数是会继续执行的，除非是0

链表怎么判断是否为empty
1. 无头节点 - head == NULL - empty
2. 有头节点 - head->next == NULL - empty
3. 有头结点且有环 - head->next == head - empty

```C++ 暴力模拟指针运动
List Insert(List L, ElementType X) {
    struct Node* New = malloc(sizeof(struct Node));
    New->Data = X;
    
    // 处理L为空的情况
    if (L->Next == NULL) {
        L->Next = New;
        return L;
    }

    List cur = L->Next;
    int cnt = -1;
    while (cur) {
    // 链表不能用cur++，这是数组连续的内存空间才能使用的，而链表是不连续的内存空间
        if (cur->Data < X) cur = cur->Next, cnt++;
        else {
            New->Next = cur;
            // 处理X小于所有数的情况，在头节点后插入
            if (cnt < 0) {
                L->Next = New;
                // return语句不能用逗号,连接
                return L;
            }
            cur = L->Next;
            while (cnt--) cur = cur->Next;
            cur->Next = New;
            return L;
        }
    }
    
    // 处理X大于所有的数的情况，在尾部插入
    cur = L->Next;
    while (cnt--) cur = cur->Next;
    cur->Next = New;
    return L;
}
```
```C++ 建立前驱指针
List Insert(List L, ElementType X) {
    struct Node* New = malloc(sizeof(struct Node));
    New->Data = X;

    // 处理L为空的情况
    if (L->Next == NULL) {
        L->Next = New;
        return L;
    }
    
    List cur = L->Next;
    List pre = L;
    // 在遍历过程中保存前驱指针
    while (cur) {
        if (cur->Data < X) {
            pre = cur;
            cur = cur->Next;
        } else {
            New->Next = cur;
            pre->Next = New;
            return L;
        }
    }
    // 处理X大于所有数的情况
    pre->Next = New;
    return L;
}
The Best:
List Insert(List L, ElementType X) {
    struct Node* New = malloc(sizeof(struct Node));
    New->Data = X;

    // 保存前驱节点
    List pre = L;
    List cur = L->Next;
    // 找到第一个Data >= X的节点
    // cur条件可以解决L为空的情况，前驱+后一个条件可以解决小于所有数和大于所有数的问题
    while (cur && cur->Data < X) {
        pre = cur;
        cur = cur->Next;
    }
    // 将新节点 插入到 pre 和 cur 中间
    New->Next = cur;
    pre->Next = New;
    return L;
}
```
