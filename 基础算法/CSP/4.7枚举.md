C++ 中的枚举（Enumeration）是一种用于定义命名常量集合的数据类型。枚举可以提高代码的可读性和可维护性，让您可以使用有意义的名称来表示特定的取值，而不必使用原始的数字常量。
```C++
// enum建立字符串常量和数字常量的对应关系
// enum 从0开始
enum Month {
    January = 1,
    February,
    March,
    April,
    May,
    June,
    July,
    Augest,
    September,
    October,
    November,
    December
};
int getDays(int year, enum Month m) {
    int n = (int)m;
    if (m == February) {
        if (isLeapYear(year)) return 29;
        return 28;
    }
    if (1 <= n && n <= 7 && n % 2 == 0) return 30;
    else if (8 <= n && n <= 12 && n % 2 == 1) return 30;
    else return 31;
}
```
