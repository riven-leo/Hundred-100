int max_len(char* s[], int n) {
    int ans = 0;
    for (int i = 0; i < n; i++) {
        int len = 0;
        // 用局部变量避免修改原变量地址
        const char* p = s[i];
        while (*p != '\0') {
            len += 1;
            p++;
        }
        ans = (ans < len ? len : ans);
    }
    return ans;
}```

3.3有序表的增删改查操作
```C++
int insert(int a[], int v) {
    for (int i = 0; i < Count; i++) {
        // 这个分支只能处理插入到中间和最开头的值
        if (a[i] < v) continue;
        else if (a[i] == v) return -1;
        else {
            // 当i为0时，j-1会越界，所以j不能=i
            for (int j = Count; j > i; j--) {
                a[j] = a[j - 1];
            }
            a[i] = v;
            Count++;
            return 1;
        }
    }
    // v大于所有a数组里的数
    a[Count] = v;
    Count++;
    return 1;
}
int del(int a[], int v) {
    for (int i = 0; i < Count; i++) {
        if (a[i] == v) {
            for (int j = i; j < Count; j++) {
                a[j] = a[j + 1];
            }
            Count--;
            return 1;
        }
    }
    return -1;
}
int modify(int a[], int v1, int v2) {
    int isv1 = 0;
    for (int i = 0; i < Count; i++) {
        if (a[i] == v2) return -1;
        if (a[i] == v1) isv1 = 1;
    }
    if (isv1) {
        int idx1 = del(a, v1);
        int idx2 = insert(a, v2);
        return 1;
    } else return -1;
}
int query(int a[], int v) {
    int l = 0, r = Count - 1;
    int mid = 0;
    // 循环条件：当l==r时有个元素没有被测试
    while (l <= r) {
        mid = (l + r) >> 1;
        if (a[mid] == v) return mid;
        // 边界更新：因为mid已经比较过不相等，所以要mid+1
        else if (a[mid] < v) l = mid + 1;
        else r = mid - 1;
    }
    return -1;
}
```

int findMin(int a[], int i, int j) {
    int min = 1e9;
    int idx = i;
    for (int k = i; k <= j; k++) {
        // 注意下标用对 这里是k不是i
        if (a[k] < min) {
            min = a[k];
            idx = k;
        }
    }
    return idx;
}```
