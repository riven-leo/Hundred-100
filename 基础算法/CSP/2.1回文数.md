3.4函数题
int findMin(int a[], int i, int j) {
    int min = 1e9;
    int idx = i;
    for (int k = i; k <= j; k++) {
        // 注意下标用对 这里是k不是i
        if (a[k] < min) {
            min = a[k];
            idx = k;
        }
    }
    return idx;
}
