/*  1. 记住0 = 48, A = 65, a = 97
    2. 控制结果在某个范围用模运算%
    3. 观察是不是还有像offset需要模运算
    4. 记住对应关系(当需要输入一整行含空格时)
        string s - getline(cin, s) - <iostream><string>
        char s[] - cin.getline(s, sizeof(s)) - <iostream>
*/
#include <iostream>
#include <string>
using namespace std;
int main() {
    string s;
    getline(cin, s);
    int offset; cin >> offset;
    // 无论是正数还是负数，1还是1000还是-1还是-27，都可以把移位数值控制在0-25
    offset = (offset % 26 + 26) % 26;
    for (char& c: s) {
        if (c == ' ') continue;
        int x = c + offset;
        // 大写在前小写在后
        // A <= c <= Z
        if (65 <= c && c <= 90) 
            // 将前置的删掉后即从A开始，同时%26控制数值在0-25
            c = (x - 65) % 26 + 'A';
        // a <= c <= z
        if (97 <= c && c <= 122) 
            c = (x - 97 ) % 26 + 'a';
    }
    cout << s << endl;
    return 0;
}
