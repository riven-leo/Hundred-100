```c++
#include <iostream>
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

// 函数：计算最长公共子序列的长度并返回具体的LCS字符串
string longestCommonSubsequence(string text1, string text2) {
    int m = text1.length();
    int n = text2.length();
    
    // dp[i][j] 表示 text1[0...i-1] 和 text2[0...j-1] 的最长公共子序列长度
    vector<vector<int>> dp(m + 1, vector<int>(n + 1, 0));
    
    // 1. 动态规划填表
    for (int i = 1; i <= m; ++i) {
        for (int j = 1; j <= n; ++j) {
            if (text1[i - 1] == text2[j - 1]) {
                dp[i][j] = dp[i - 1][j - 1] + 1;
            } else {
                dp[i][j] = max(dp[i - 1][j], dp[i][j - 1]);
            }
        }
    }
    
    // 2. 反向回溯构造最长公共子序列
    string lcs = "";
    int i = m, j = n;
    while (i > 0 && j > 0) {
        if (text1[i - 1] == text2[j - 1]) {
            lcs.push_back(text1[i - 1]);
            i--;
            j--;
        } else if (dp[i - 1][j] > dp[i][j - 1]) {
            i--;
        } else {
            j--;
        }
    }
    reverse(lcs.begin(), lcs.end()); // 构造出的顺序是反的，需要翻转
    return lcs;
}

int main() {
    string text1 = "abcde";
    string text2 = "ace";
    
    string result = longestCommonSubsequence(text1, text2);
    cout << "最长公共子序列: " << result << endl;
    cout << "长度: " << result.length() << endl;
    
    return 0;
}
```
