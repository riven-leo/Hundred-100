#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, W;
    cin >> n >> W;

    // 下标从 1 开始，方便处理 i - 1 的边界情况
    vector<int> w(n + 1), v(n + 1);
    for (int i = 1; i <= n; ++i) {
        cin >> w[i] >> v[i];
    }

    // dp[i][j] 表示前 i 件物品放入容量为 j 的背包中的最大价值
    vector<vector<int>> dp(n + 1, vector<int>(W + 1, 0));

    for (int i = 1; i <= n; ++i) {
        for (int j = 0; j <= W; ++j) {
            if (j < w[i]) {
                // 当前背包容量装不下第 i 件物品，只能继承前 i-1 件的结果
                dp[i][j] = dp[i - 1][j];
            } else {
                // 在“不放”和“放”第 i 件物品之间取最大值
                dp[i][j] = max(dp[i - 1][j], dp[i - 1][j - w[i]] + v[i]);
            }
        }
    }

    cout << dp[n][W] << endl;
    return 0;
}

#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int knapsack1D(int W, const vector<int>& weight, const vector<int>& value, int n) {
    vector<int> dp(W + 1, 0);
    for (int i = 0; i < n; ++i) {
        for (int j = W; j >= weight[i]; --j) {
            dp[j] = max(dp[j], dp[j - weight[i]] + value[i]);
        }
    }
    return dp[W];
}
