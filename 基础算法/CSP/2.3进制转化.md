// 针对2、8、16,由短除法后的余数，进行对应字符的转换
char toc(int n) {
    char c;
    if (0 <= n && n <= 9) c = (char)('0' + n);
    if (10 <= n && n <= 15) c = (char)(n - 10 + 'A');
    return c;
}
// 10进制转x进制的
string tox(int n, int x) {
    string ans = "";
    while (n != 0) {
        ans += toc(n % x);
        n /= x;
    }
    // reverse ans
    // 1.双指针法，左指针从0开始，右指针从末尾size()-1开始，交换一对，左++，右--，当左指针和右指针重合的时候结束
    int l = 0, r = ans.size() - 1;
    while (l < r) {
        char t;
        t = ans[l];
        ans[l] = ans[r];
        ans[r] = t;
        l++, r--;
    }

    return ans;
}
