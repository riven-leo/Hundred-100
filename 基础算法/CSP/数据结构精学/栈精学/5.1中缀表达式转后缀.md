这道题很多问题，但是大部分是数据类型和括号位置问题

```C++
#include <iostream>
#include <stack>
#include <string>
using namespace std;

int priority (char op) {
    if (op == '*' || op == '/') return 2;
    else if (op == '+' || op == '-') return 1;
    else return 0;
}

int main() {
    // 运算符栈
    stack<char> v;
    string s; cin >> s;
    int n = 1;
    // 第一个有可能是括号
    if (s[0] == '(') {
        v.push(s[0]);
        cout << s[1];
        n = 2;
    }
    else cout << s[0];
    for (int i = n; i < s.size(); i++) {
        // 数字，直接输出
        if ('0' <= s[i] && s[i] <= '9') cout << ' ' << s[i];
        // 右括号，则栈弹元素到左括号
        else if (s[i] == ')') {
            while (v.top() != '(') {
                cout << ' ' << v.top();
                v.pop();
            }
            v.pop();
        } 
        else {
            // 单独处理'('
            if (s[i] == '(') { v.push(s[i]); continue; }
            // 栈非空且栈顶非左括号的情况下，v.top()优先级 >= s[i]优先级，则弹出再压入
            while (!v.empty() && v.top() != '(' && priority(v.top()) >= priority(s[i])) {
                cout << ' ' << v.top();
                v.pop();
            }
            v.push(s[i]);
        }
    }
    while (!v.empty()) cout << ' ' << v.top(), v.pop();
    return 0;
}
```
