银行排队算法之单队列多窗口问题：
1. 初始化：读入所有顾客的信息，并按到达时间升序排序。
2. 遍历顾客：对于每一个顾客，寻找当前最早空闲的窗口（即 Free Time 最小的窗口）。
3. 计算等待与服务时间：
	• 情况 A（窗口在等顾客）：如果该窗口的 Free Time ≤ 顾客的到达时间，说明顾客无需等待，直接办理。更新该窗口的 Free Time = 顾客到达时间 + 处理时间。
	• 情况 B（顾客在等窗口）：如果该窗口的 Free Time > 顾客的到达时间，说明顾客需要等待。等待时间 = Free Time - 顾客到达时间。累加等待时间后，更新该窗口的 Free Time = 原 Free Time + 处理时间。
4. 统计结果：遍历结束后，计算总等待时间 / 顾客总数，得到平均等待时间；并找出 Free Time 的最大值或最长等待时间。

该题的小变化是：当窗口有多个空闲时，顾客选择编号更小的窗口，否则选择空闲时间最早的窗口


```C++
#include <iostream>
#include <vector>
#include <queue>
#include <utility> // 工具头文件 pair在内
#include <algorithm> // minmax在内
using namespace std;
typedef pair<int, int> PII;
int main() {
    int n; cin >> n;
    int N = n;
    queue<PII> q; // arrivetime solvetime
    PII tp;
    while (N--) {
        int T, P; cin >> T >> P; 
        if (P > 60) P = 60;
        tp = make_pair(T, P);
        q.push(tp);
    }
    int K; cin >> K;
    vector<int> w(K, 0); // freetime
    vector<int> cnt(K, 0);

    double total_wait_t = 0;
    int max_wait_t = 0;
    // 遍历排队的人
    for (int i = 0; i < n; i++) {
        PII customer = q.front();
        q.pop();

        int arrive = customer.first;
        int solve = customer.second;

        // 当有多个窗口可选择时，假设顾客总是选择编号最小的窗口。
        // 优先选择编号更小的空闲窗口
        int mfree = -1;
        for (int j = 0; j < K; j++) {
            if (w[j] <= arrive) {
                mfree = j;
                break;
            }
        }
        // 如果没有空闲窗口，选最早空闲的窗口
        if (mfree == -1) {
            mfree = 0;
            for (int j = 1; j < K; j++) {
                if (w[j] < w[mfree]) mfree = j;
            }
        }
        // 核心逻辑：窗口freetime <= 客户arrivetime，则客户不用等，更新该窗口freetime
        if (w[mfree] <= arrive) {
            w[mfree] = arrive + solve;
            cnt[mfree]++;
        }
        // 反之客户要等待freetime-arrivetime时间，再更新freetime
        else {
            int wait_t = w[mfree] - arrive;
            total_wait_t += wait_t;
            max_wait_t = max(max_wait_t, wait_t);
            w[mfree] += solve;
            cnt[mfree]++;
        }
    }
    int last = 0;
    for(auto x: w) last = max(last, x);
    printf("%.1f", total_wait_t / n);
    cout << " " << max_wait_t << " " << last << endl;
    int f = 1;
    for (int i = 0; i < K; i++) {
        if (f) cout << cnt[i], f = 0;
        else cout << " " << cnt[i];
    }
    cout << endl;
    return 0;
}```
