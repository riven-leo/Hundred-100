约瑟夫环用queue模拟循环出队

```cpp
int n, m; cin >> n >> m;
    queue<int> q;
    for (int i = 1; i <= n; i++) {
        q.push(i);
    }
    while (!q.empty()) {
        for (int i = 1; i < m; i++) {
            q.push(q.front());
            q.pop();
        }
        cout << q.front() << endl;
        q.pop();
    }
```
