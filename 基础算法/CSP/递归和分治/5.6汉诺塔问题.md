算法原理：
假设要把 n 个盘子从 A 柱借由 B 柱移到 C 柱，可以把复杂的大问题拆解为 3 个步骤：
1. 将上面 n-1 个盘子，从 A 柱借助 C 柱移动到 B 柱。
2. 将 A 柱底部剩下第 n 个（最大的）盘子，直接移动到 C 柱。
3. 将 B 柱上的 n-1 个盘子，借助 A 柱移动到 C 柱。
```C++
#include <iostream>
using namespace std;

/**
 * @param n     圆盘的数量
 * @param from  起始柱子（如 'A'）
 * @param aux   辅助柱子（如 'B'）
 * @param to    目标柱子（如 'C'）
 */
void hanoi(int n, char from, char aux, char to) {
    if (n == 1) {
        // 递归终止条件：只有一个盘子时，直接从起始柱移到目标柱
        cout << "Move disk 1 from " << from << " -> " << to << endl;
        return;
    }

    // 1. 将上面 n-1 个盘子从 from 借助 to 移动到 aux
    hanoi(n - 1, from, to, aux);

    // 2. 将第 n 个（最底下的）盘子从 from 移动到 to
    cout << "Move disk " << n << " from " << from << " -> " << to << endl;

    // 3. 将 aux 上的 n-1 个盘子借助 from 移动到 to
    hanoi(n - 1, aux, from, to);
}

int main() {
    int n = 3; // 盘子的数量
    cout << "Hanoi steps for " << n << " disks:" << endl;
    hanoi(n, 'A', 'B', 'C');
    return 0;
}
```
