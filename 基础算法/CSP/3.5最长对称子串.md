最长回文子串解法
• 对基本数据类型：如 sizeof(int) 通常返回 4。
• 对静态数组：如 int arr[10]，sizeof(arr) 返回整个数组占用的字节数（\(10 \times 4 = 40\) 字节）。
```c++
/*  记住区别：获取实际长度
    1. string s - s.size() <string>
    2. char s[] - strlen(s) <string.h> 且普通数组进入函数会退化成指针，大小要另外传入
    上述函数都是获取实际长度，而char数组型大小可能是s[1000]，但是实际长度可能只有15
*/
// 扩展算法
int expand(const string& s, int l, int r) {
    while (l >= 0 && r < s.size() && s[l] == s[r]) {
        l--, r++;
    }
    return r - l - 1;
}

int main() {
    string s;
    getline(cin, s);
    int ans = 0;
    // 遍历每一个位置，将其当成中心寻找回文串
    for (int i = 0; i < s.size(); i++) {
        // 奇数中心扩展 aba，偶数中心扩展abba
        int odd = expand(s, i, i);
        int even = expand(s, i, i + 1);
        ans = max(ans, max(odd, even));
    }
    cout << ans << endl;
    return 0;
}
```
