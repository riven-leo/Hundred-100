| 操作                  | 给的信息         | 想快速得到什么         |
| ------------------- | ------------ | --------------- |
| `search(movie)`     | movie        | 找这个电影最便宜的 5 个商店 |
| `rent(shop, movie)` | shop + movie | 修改这份库存状态        |
| `drop(shop, movie)` | shop + movie | 修改这份库存状态        |
| `report()`          | 无参数          | 找所有已出租中最便宜的 5 个 |

设计题想数据结构

class MovieRentingSystem {
    set<tuple<int, int, int>> rented;
    unordered_map<long long, int> smtop; //
    unordered_map<int, set<pair<int, int>>> unrented;

public:
    MovieRentingSystem(int n, vector<vector<int>>& entries) {
        for (auto& e: entries) {
            int shop = e[0], movie = e[1], price = e[2];
            unrented[movie].emplace(price, shop); //
            smtop[1LL * shop << 32 | movie] = price; //
        }
    }
    
    vector<int> search(int movie) {
        auto it = unrented.find(movie);
        if (it == unrented.end()) return {};
        vector<int> ans;
        for (auto& [_, shop]: it->second) {
            ans.push_back(shop);
            if (ans.size() == 5) break;
        }
        return ans;
    }
    
    void rent(int shop, int movie) {
        int price = smtop[1LL * shop << 32 | movie];

        unrented[movie].erase({shop, price});
        rented.emplace(price, shop, movie);
    }
    
    void drop(int shop, int movie) {
        int price = smtop[1LL * shop << 32 | movie];

        rented.erase({price, shop, movie});
        unrented[movie].emplace(price, shop);
    }
    
    vector<vector<int>> report() {
        vector<vector<int>> ans;
        for (auto& [_, shop, movie]: rented) {
            ans.push_back({shop, movie});
            if (ans.size() == 5) break;
        }
        return ans;
    }
};
