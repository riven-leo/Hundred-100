1. 以下程序的输出结果是什么？

```c

    int a = 5, b = 2;
    float c = (float)a / b;
    printf("%.2f\n", c);
    return 0;
}
```

**答案**：`2.50`。这里先将`a`显式转换为`float`类型，再进行除法运算，得到结果`2.5`，按照`%.2f`格式输出保留两位小数。

1. 分析表达式`3 + 4 * 2 < 10 || 5 > 3 && 2 < 1`的求值过程和结果。

**答案**：先计算乘法`4 * 2 = 8`，然后计算`3 + 8 = 11`，接着比较`11 < 10`为假，再计算`5 > 3`为真，`2 < 1`为假，最后计算逻辑与`5 > 3 && 2 < 1`为假，最后计算逻辑或`3 + 4 * 2 < 10 || 5 > 3 && 2 < 1`为假。结果为`0`。

// 冒泡排序函数
void bubble_sort(int arr[], int n) {
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                int temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }
}

    int n = sizeof(arr) / sizeof(arr[0]);
    bubble_sort(arr, n);
    printf("排序后的数组：");

# 九、链表

## 重点

- **链表的概念与结构**：理解链表是由一系列节点组成的数据结构，每个节点包含数据域和指针域，指针域指向下一个节点，通过节点之间的指针连接形成链状结构。例如，定义一个简单的单向链表节点结构体：

```c
typedef struct Node {
    int data;
    struct Node *next;
} Node;
```

- **链表的基本操作实现原理**：
- **创建链表**：掌握头插法和尾插法创建链表的过程。头插法是每次将新节点插入到链表头部，尾插法是将新节点插入到链表尾部。
- **遍历链表**：能够使用循环和指针移动依次访问链表中的每个节点，输出节点数据或进行其他操作。
- **插入节点**：实现在链表指定位置插入新节点，需要正确调整插入位置前后节点的指针指向，以保持链表的连续性。
- **删除节点**：找到要删除的节点，并正确处理其前后节点的指针关系，释放被删除节点的内存空间。

## 难点

- **指针操作的复杂性与正确性**：链表操作中大量涉及指针的操作，指针的赋值、移动和指向关系的调整容易出错，尤其是在处理边界情况（如空链表、只有一个节点的链表、在链表头部或尾部插入/删除节点等）时，需要特别小心以确保指针操作的正确性，避免出现野指针或内存泄漏等问题。
- **理解链表的动态特性与内存管理**：链表的长度可以动态变化，在插入和删除节点时涉及到内存的动态分配和释放。需要深入理解这些操作对内存的影响，以及如何正确地管理内存，确保程序的稳定性和效率。例如，在删除节点时，要记得释放被删除节点所占用的内存空间，防止内存泄漏；在创建新节点时，要确保内存分配成功。

## 例题

例题 1：使用头插法创建链表并遍历输出

```c
#include <stdio.h>
#include <stdlib.h>

// 链表节点结构体
typedef struct Node {
    int data;
    struct Node *next;
} Node;

// 使用头插法创建链表
Node *create_linked_list_head_insert(int arr[], int n) {
    Node *head = NULL;
    for (int i = 0; i < n; i++) {
        // 创建新节点
        Node *new_node = (Node *)malloc(sizeof(Node));
        new_node->data = arr[i];
        // 将新节点插入到头部
        new_node->next = head;
        head = new_node;
    }
    return head;
}

// 遍历链表并输出节点数据
void print_linked_list(Node *head) {
    Node *current = head;
    while (current!= NULL) {
        printf("%d ", current->data);
        current = current->next;
    }
    printf("\n");
}

int main() {
    int arr[] = {1, 3, 5, 7, 9};
    int n = sizeof(arr) / sizeof(arr[0]);

    Node *head = create_linked_list_head_insert(arr, n);
    print_linked_list(head);

    // 释放链表内存
    Node *temp;
    while (head!= NULL) {
        temp = head;
        head = head->next;
        free(temp);
    }

    return 0;
}
```

**解析**：在 `create_linked_list_head_insert` 函数中，首先初始化头指针 `head` 为 `NULL`。然后遍历数组，对于每个数组元素，创建一个新节点。新节点的 `next` 指针先指向当前的头节点（即原来链表的头部），然后将新节点更新为链表的头部。这样就实现了头插法创建链表。在 `print_linked_list` 函数中，通过一个循环，从链表头部开始，依次输出每个节点的数据，直到到达链表末尾（`current` 指针为 `NULL`）。最后在 `main` 函数中，创建链表并输出后，需要释放链表所占用的内存，通过一个循环，依次释放每个节点的内存空间。

例题 2：在有序链表中插入一个节点，使链表仍然有序

```c
#include <stdio.h>
#include <stdlib.h>

// 链表节点结构体
typedef struct Node {
    int data;
    struct Node *next;
} Node;

// 插入节点到有序链表
Node *insert_into_sorted_linked_list(Node *head, int value) {
    Node *new_node = (Node *)malloc(sizeof(Node));
    new_node->data = value;
    new_node->next = NULL;

    // 如果链表为空或者新节点的值小于头节点的值，将新节点插入到头部
    if (head == NULL || value < head->data) {
        new_node->next = head;
        return new_node;
    }

    Node *current = head;
    // 找到插入位置
    while (current->next!= NULL && current->next->data < value) {
        current = current->next;
    }
    // 插入新节点
    new_node->next = current->next;
    current->next = new_node;

    return head;
}

// 遍历链表并输出节点数据
void print_linked_list(Node *head) {
    Node *current = head;
    while (current!= NULL) {
        printf("%d ", current->data);
        current = current->next;
    }
    printf("\n");
}

int main() {
    Node *head = NULL;
    head = insert_into_sorted_linked_list(head, 5);
    head = insert_into_sorted_linked_list(head, 3);
    head = insert_into_sorted_linked_list(head, 8);
    head = insert_into_sorted_linked_list(head, 1);

    print_linked_list(head);

    // 释放链表内存
    Node *temp;
    while (head!= NULL) {
        temp = head;
        head = head->next;
        free(temp);
    }

    return 0;
}
```

**解析**：`insert_into_sorted_linked_list` 函数用于在有序链表中插入新节点。首先创建新节点并初始化。然后判断链表是否为空或者新节点的值小于头节点的值，如果是则将新节点插入到头部。接着通过循环找到合适的插入位置，即当前节点的下一个节点的值大于新节点的值的位置。最后将新节点插入到该位置，调整指针关系。在 `main` 函数中，依次插入几个节点并输出链表，最后释放链表内存。

例题 3：删除链表中指定值的节点

```c
#include <stdio.h>
#include <stdlib.h>

// 链表节点结构体
typedef struct Node {
    int data;
    struct Node *next;
} Node;

// 删除链表中指定值的节点
Node *delete_node(Node *head, int value) {
    Node *prev = NULL;
    Node *current = head;

    // 处理头节点是要删除的节点的情况
    while (current!= NULL && current->data == value) {
        head = current->next;
        free(current);
        current = head;
    }

    // 处理中间节点和尾节点
    while (current!= NULL) {
        while (current!= NULL && current->data!= value) {
            prev = current;
            current = current->next;
        }
        if (current == NULL) {
            break;
        }
        prev->next = current->next;
        free(current);
        current = prev->next;
    }

    return head;
}

// 遍历链表并输出节点数据
void print_linked_list(Node *head) {
    Node *current = head;
    while (current!= NULL) {
        printf("%d ", current->data);
        current = current->next;
    }
    printf("\n");
}

int main() {
    Node *head = NULL;
    Node *node1 = (Node *)malloc(sizeof(Node));
    node1->data = 1;
    head = node1;
    Node *node2 = (Node *)malloc(sizeof(Node));
    node2->data = 2;
    node1->next = node2;
    Node *node3 = (Node *)malloc(sizeof(Node));
    node3->data = 3;
    node2->next = node3;
    Node *node4 = (Node *)malloc(sizeof(Node));
    node4->data = 2;
    node3->next = node4;
    node4->next = NULL;

    head = delete_node(head, 2);

    print_linked_list(head);

    // 释放链表内存
    Node *temp;
    while (head!= NULL) {
        temp = head;
        head = head->next;
        free(temp);
    }

    return 0;
}
```

**解析**：`delete_node` 函数用于删除链表中指定值的节点。首先处理头节点是要删除节点的情况，通过循环找到并释放头节点，更新头指针。然后处理中间节点和尾节点，通过两个嵌套循环，外层循环用于遍历链表，内层循环用于找到要删除的节点。找到后调整指针关系，释放节点内存，并更新当前节点指针。在 `main` 函数中，创建一个包含几个节点的链表，删除指定值的节点后输出链表，最后释放链表内存。

# 十、文件操作

## 重点

- 文件指针的概念与使用：`FILE` 类型指针用于指向打开的文件，通过它来操作文件。例如：`FILE *fp;` 声明了一个文件指针变量 `fp`。
- 文件的打开与关闭：使用 `fopen` 函数打开文件，其原型为 `FILE *fopen(const char *filename, const char *mode)`，其中 `filename` 是文件名，`mode` 是打开模式（如 “r” 只读、“w” 只写、“a” 追加等）。使用 `fclose` 函数关闭文件，如 `fclose(fp);` 关闭由 `fp` 指向的文件，确保资源正确释放并将缓冲区数据写入文件。
- 文件的顺序读写：
- 字符读写：`fgetc` 从文件读取一个字符，`fputc` 向文件写入一个字符。例如：`char ch = fgetc(fp);` 从 `fp` 指向的文件读取一个字符到 `ch` 变量，`fputc('A', fp);` 向 `fp` 指向的文件写入字符 `'A'`。
- 字符串读写：`fgets` 从文件读取一行字符串，`fputs` 向文件写入字符串。如 `fgets(str, sizeof(str), fp);` 从 `fp` 读取一行到 `str` 数组中，`fputs("Hello", fp);` 向 `fp` 指向的文件写入 “Hello”。
- 格式化读写：`fscanf` 和 `fprintf` 类似于 `scanf` 和 `printf`，但用于文件操作。例如：`fscanf(fp, "%d", &num);` 从文件读取一个整数到 `num` 变量，`fprintf(fp, "%.2f", 3.14);` 向文件写入格式化后的浮点数。
- 文件的随机读写：通过 `fseek` 函数移动文件指针位置，`ftell` 函数获取当前文件指针位置，`rewind` 函数将文件指针移回文件开头。例如：`fseek(fp, 10L, SEEK_SET);` 将文件指针从文件开头向后移动 10 个字节，`long pos = ftell(fp);` 获取当前文件指针位置，`rewind(fp);` 使文件指针回到文件开头。

## 难点

- 正确选择文件打开模式：不同的打开模式决定了对文件的操作权限和文件的初始状态，若模式选择错误可能导致文件操作失败或数据损坏。例如，以 “w” 模式打开已存在的文件会清空文件内容。
- 处理文件读写错误：文件读写过程中可能因各种原因出错，如文件不存在、磁盘空间不足等。需要正确判断 `fopen` 等函数的返回值以及使用 `ferror` 函数检查读写错误，并进行相应的错误处理。
- 理解文件指针的移动和随机读写的逻辑：在随机读写时，准确计算偏移量和理解不同的移动基准（`SEEK_SET` 开头、`SEEK_CUR` 当前位置、`SEEK_END` 结尾）需要清晰的逻辑思维，否则容易导致读取或写入位置错误。

## 例题

1. 从一个文本文件 `input.txt` 中读取所有字符并输出到控制台，统计字符个数。

```c
#include <stdio.h>

int main() {
    FILE *fp;
    char ch;
    int count = 0;

    fp = fopen("input.txt", "r");
    if (fp == NULL) {
        printf("文件打开失败！\n");
        return 1;
    }

    while ((ch = fgetc(fp))!= EOF) {
        putchar(ch);
        count++;
    }

    fclose(fp);
    printf("\n 文件中的字符个数为：%d\n", count);
    return 0;
}
```

1. 将一组整数写入到文件 `data.bin` 中，然后从文件中读取并输出到控制台。

```c
#include <stdio.h>

int main() {
    FILE *fp;
    int arr[] = {1, 2, 3, 4, 5};
    int num, i;

    // 写入文件
    fp = fopen("data.bin", "wb");
    if (fp == NULL) {
        printf("文件打开失败！\n");
        return 1;
    }
    for (i = 0; i < sizeof(arr) / sizeof(arr[0]); i++) {
        fwrite(&arr[i], sizeof(int), 1, fp);
    }
    fclose(fp);

    // 读取文件
    fp = fopen("data.bin", "rb");
    if (fp == NULL) {
        printf("文件打开失败！\n");
        return 1;
    }
    while (fread(&num, sizeof(int), 1, fp) == 1) {
        printf("%d ", num);
    }
    fclose(fp);

    return 0;
}
```

1. 复制一个文本文件 `source.txt` 到 `destination.txt`。

```c
#include <stdio.h>

int main() {
    FILE *source, *destination;
    char buffer[1024];
    size_t bytesRead;

    source = fopen("source.txt", "r");
    if (source == NULL) {
        printf("源文件打开失败！\n");
        return 1;
    }

    destination = fopen("destination.txt", "w");
    if (destination == NULL) {
        printf("目标文件打开失败！\n");
        fclose(source);
        return 1;
    }

    while ((bytesRead = fread(buffer, 1, sizeof(buffer), source)) > 0) {
        fwrite(buffer, 1, bytesRead, destination);
    }

    fclose(source);
    fclose(destination);

    return 0;
}
```
